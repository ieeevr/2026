document.addEventListener('DOMContentLoaded', function() {
    var buttons = document.querySelectorAll('.btn-info');
    buttons.forEach(function(btn) {
        btn.addEventListener('click', function() {
            var name = this.getAttribute('data-name') || '';
            var items = this.getAttribute('data-items') || '';
            var web = this.getAttribute('data-website') || '';
            var info = this.getAttribute('data-info') || '';
            
            document.getElementById('mName').innerText = name;
            
            var itemsWrapper = document.getElementById('mItemsWrapper');
            if(items) {
                document.getElementById('mItems').innerText = items;
                itemsWrapper.style.display = 'block';
            } else {
                itemsWrapper.style.display = 'none';
            }
            
            var webWrapper = document.getElementById('mWebWrapper');
            if(web) {
                var webLink = document.getElementById('mWeb');
                webLink.href = web.startsWith('http') ? web : 'http://' + web;
                webLink.innerText = web;
                webWrapper.style.display = 'block';
            } else {
                webWrapper.style.display = 'none';
            }
            
            document.getElementById('mInfo').innerText = info;
            document.getElementById('companyModal').style.display = 'flex';
        });
    });

    document.getElementById('companyModal').addEventListener('click', function(e) {
        if (e.target === this) {
            closeModal();
        }
    });
});

function closeModal() {
    document.getElementById('companyModal').style.display = 'none';
}
</script>
